# Builds the UGens of isospin as a loadable SuperCollider plugin, for scsynth
# and for supernova, from the output of scripts/build-sc-static.py. The
# install step writes a SuperCollider extension folder:
#
#   <prefix>/isospin/plugins/      isospin and isospin_supernova
#   <prefix>/isospin/Classes/      the class files
#   <prefix>/isospin/HelpSource/   the help pages
#   <prefix>/isospin/              README.md, LICENSE, NOTICE.md, BUILDINFO
#
# scripts/build-sc-plugin.py runs this project. The settings:
#
#   ISOSPIN_SC_STATIC  The folder of a static build. It has isospin.cmake.
#   SC_INCLUDE         The include folder of the SuperCollider source, with
#                      plugin_interface/, common/ and server/.
#   ISOSPIN_SUPERNOVA  Also build the supernova plugin (default ON).

cmake_minimum_required(VERSION 3.16)
project(isospin_sc_plugin LANGUAGES CXX)

set(ISOSPIN_SC_STATIC "" CACHE PATH "The folder of a static build (output of build-sc-static.py)")
set(SC_INCLUDE "" CACHE PATH "The include folder of the SuperCollider source")
option(ISOSPIN_SUPERNOVA "Also build the supernova plugin" ON)

if(NOT EXISTS "${ISOSPIN_SC_STATIC}/isospin.cmake")
    message(FATAL_ERROR "ISOSPIN_SC_STATIC must be a static build folder with isospin.cmake")
endif()
if(NOT EXISTS "${SC_INCLUDE}/plugin_interface/SC_PlugIn.h")
    message(FATAL_ERROR "SC_INCLUDE must be the include folder of the SuperCollider source")
endif()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Only the entry points of PluginLoad() are visible outside the plugin.
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

# The source list and the per-source settings (ISOSPIN_SC_STATIC_SOURCES).
include("${ISOSPIN_SC_STATIC}/isospin.cmake")

# The entry point of the plugin. It calls the load function of each UGen.
set(ISOSPIN_LOAD_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/isospin_load.cpp")
file(WRITE "${ISOSPIN_LOAD_SOURCE}.new" [=[
// Generated by isospin sc-plugin/CMakeLists.txt. Do not edit.
// The entry point of the loadable plugin with all UGens of isospin.
#include <SC_PlugIn.h>

#include "isospin_sc_static.h"

PluginLoad(Isospin) { Isospin_Load(inTable); }
]=])
configure_file("${ISOSPIN_LOAD_SOURCE}.new" "${ISOSPIN_LOAD_SOURCE}" COPYONLY)

# SuperCollider looks for .so on Linux and .scx on macOS and Windows.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(ISOSPIN_PLUGIN_SUFFIX ".so")
else()
    set(ISOSPIN_PLUGIN_SUFFIX ".scx")
endif()

function(isospin_plugin target)
    add_library(${target} MODULE ${ISOSPIN_SC_STATIC_SOURCES} "${ISOSPIN_LOAD_SOURCE}")
    target_include_directories(${target} SYSTEM PRIVATE
        "${SC_INCLUDE}/plugin_interface" "${SC_INCLUDE}/common" "${SC_INCLUDE}/server")
    target_include_directories(${target} PRIVATE "${ISOSPIN_SC_STATIC}")
    if(WIN32)
        target_compile_definitions(${target} PRIVATE NOMINMAX _USE_MATH_DEFINES)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
    set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX "${ISOSPIN_PLUGIN_SUFFIX}")
    install(TARGETS ${target} LIBRARY DESTINATION isospin/plugins)
endfunction()

isospin_plugin(isospin)
if(ISOSPIN_SUPERNOVA)
    isospin_plugin(isospin_supernova)
    target_compile_definitions(isospin_supernova PRIVATE SUPERNOVA)
endif()

install(DIRECTORY "${ISOSPIN_SC_STATIC}/classes/" DESTINATION isospin/Classes)
install(DIRECTORY "${ISOSPIN_SC_STATIC}/help/" DESTINATION isospin/HelpSource/Classes)
install(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/README.md"
    "${ISOSPIN_SC_STATIC}/LICENSE"
    "${ISOSPIN_SC_STATIC}/NOTICE.md"
    "${ISOSPIN_SC_STATIC}/BUILDINFO"
    DESTINATION isospin)
